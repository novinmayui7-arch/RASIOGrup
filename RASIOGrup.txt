        /* ================= STATE & AUDIO MANAGEMENT ================= */
        const gameState = {
            groupName: '',

            currentLevelId: null,
            audioMuted: false,
            levelsProgress: {
                1: { completed: false, puzzleSolved: false },

        let currentSlotAnswers = {};

        // Web Audio Synthesizer
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playSound(type) {
            if (gameState.audioMuted) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                const now = audioCtx.currentTime;
                if (type === 'click') {
                    osc.frequency.setValueAtTime(400, now);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'correct') {
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.setValueAtTime(659.25, now + 0.08);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
                    osc.start(now);
                    osc.stop(now + 0.25);
                } else if (type === 'flip') {
                    osc.frequency.setValueAtTime(300, now);
                    osc.frequency.exponentialRampToValueAtTime(800, now + 0.15);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
                    osc.start(now);
                    osc.stop(now + 0.15);
                } else if (type === 'win') {
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.setValueAtTime(659.25, now + 0.1);
                    osc.frequency.setValueAtTime(783.99, now + 0.2);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                }
            } catch (e) { console.log(e); }
        }

            board.innerHTML = '';
            tray.innerHTML = '';

            lvl.puzzleSlots.forEach((slot) => {
                const slotDiv = document.createElement('div');
                slotDiv.className = "bg-white/90 rounded-xl border-2 border-slate-200 p-2.5 flex flex-col justify-between items-center text-center relative transition shadow-sm hover:border-palu-teal cursor-pointer select-none";

                        [ Kosong ]
                    </div>
                `;

                slotDiv.onclick = () => handleSlotClick(slot.id);
                slotDiv.ondragover = (e) => e.preventDefault();
                slotDiv.ondrop = (e) => handleDropOnSlot(e, slot.id);

                board.appendChild(slotDiv);
            });

            const shuffledPieces = [...lvl.puzzleSlots].sort(() => Math.random() - 0.5);

            shuffledPieces.forEach((piece) => {
                const pieceDiv = document.createElement('div');
                pieceDiv.id = `piece_${piece.id}`;
                pieceDiv.draggable = true;
                pieceDiv.className = "bg-gradient-to-r from-palu-teal to-teal-600 text-white font-heading font-bold text-xs sm:text-sm p-3 rounded-xl shadow-md hover:shadow-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-center transition transform hover:-translate-y-0.5 select-none";
                pieceDiv.innerText = piece.answer;
                pieceDiv.dataset.pieceId = piece.id;

                pieceDiv.ondragstart = (e) => {
                    e.dataTransfer.setData('text/plain', piece.id);
                    pieceDiv.classList.add('dragging');
                };
                pieceDiv.ondragend = () => pieceDiv.classList.remove('dragging');
                pieceDiv.onclick = () => handlePieceClick(piece.id);

                tray.appendChild(pieceDiv);
            });

            updateRemainingCount();
        }

        function handlePieceClick(pieceId) {
            playSound('click');
            const pieceDiv = document.getElementById(`piece_${pieceId}`);
            if (selectedPieceId === pieceId) {
                selectedPieceId = null;
                pieceDiv.classList.remove('ring-4', 'ring-amber-400');
            } else {
                if (selectedPieceId !== null) {
                    const prev = document.getElementById(`piece_${selectedPieceId}`);
                    if (prev) prev.classList.remove('ring-4', 'ring-amber-400');
                }
                selectedPieceId = pieceId;
                pieceDiv.classList.add('ring-4', 'ring-amber-400');
            }
        }

        function handleSlotClick(slotId) {
            if (selectedPieceId !== null) {
                placePieceInSlot(selectedPieceId, slotId);
                selectedPieceId = null;
            }
        }

        function handleDropOnSlot(e, slotId) {
            e.preventDefault();
            const pieceId = parseInt(e.dataTransfer.getData('text/plain'));
            if (!isNaN(pieceId)) {
                placePieceInSlot(pieceId, slotId);
            }
        }

        function placePieceInSlot(pieceId, slotId) {
            if (pieceId === slotId) {
                playSound('correct');
                currentSlotAnswers[slotId] = pieceId;

                const lvl = levelsData.find(l => l.id === gameState.currentLevelId);
                const pieceObj = lvl.puzzleSlots.find(p => p.id === pieceId);

                const slotContent = document.getElementById(`slotContent_${slotId}`);
                slotContent.className = "w-full py-1.5 px-2 bg-emerald-500 text-white rounded-lg text-xs font-black border border-emerald-600 shadow-sm animate-bounce";
                slotContent.innerText = pieceObj.answer;

                const pieceDiv = document.getElementById(`piece_${pieceId}`);
                if (pieceDiv) pieceDiv.remove();

                updateRemainingCount();
                checkPuzzleCompletion();
            } else {
                playSound('click');
                alert('Potongan jawaban belum tepat untuk soal ini. Coba lagi!');
            }
        }

        function updateRemainingCount() {
            const count = 4 - Object.keys(currentSlotAnswers).length;
            document.getElementById('remainingPiecesText').innerText = `${count} Tersisa`;
        }

        function handlePuzzleSuccess() {
            playSound('flip');
            gameState.levelsProgress[gameState.currentLevelId].puzzleSolved = true;
            
            document.getElementById('cardFlipper').classList.add('rotate-y-180');
            document.getElementById('gameStageIndicator').innerText = "Fase 2: Soal HOTS & LKPD";
        }

        function flipBackToPuzzle() {
            playSound('flip');
            document.getElementById('cardFlipper').classList.remove('rotate-y-180');
            document.getElementById('gameStageIndicator').innerText = "Fase 1: Puzzle Kerja Sama";
        }

        function setupHotsStage(lvl) {
            document.getElementById('hotsLandmarkSubtitle').innerText = lvl.landmark;
            document.getElementById('hotsStoryText').innerText = lvl.hots.story;
            document.getElementById('hotsLandmarkSvgPreview').innerHTML = landmarkSvgs[lvl.svgKey];
        }

        function markLevelCompleteAndReturn() {
            playSound('win');
            gameState.levelsProgress[gameState.currentLevelId].completed = true;
            goToLevelSelect();
        }

        function showReportModal() {
            playSound('click');
            document.getElementById('reportGroupName').innerText = gameState.groupName || '-';
            document.getElementById('reportGroupMembers').innerText = gameState.members.join(', ') || '-';

            let completedCount = 0;
            const detailsContainer = document.getElementById('reportLevelDetails');
            detailsContainer.innerHTML = '';

            levelsData.forEach(lvl => {
                const prog = gameState.levelsProgress[lvl.id];
                if (prog.completed) completedCount++;

                const item = document.createElement('div');
                item.className = "p-4 rounded-xl border border-slate-200 bg-slate-50 space-y-2";
                item.innerHTML = `
                    <div class="flex items-center justify-between border-b border-slate-200 pb-2">
                        <span class="font-extrabold text-xs text-slate-800">Level ${lvl.id}: ${lvl.landmark} (${lvl.title})</span>
                        <span class="text-[10px] font-bold px-2 py-0.5 rounded-full ${prog.completed ? 'bg-emerald-100 text-emerald-800' : 'bg-slate-200 text-slate-600'}">
                            ${prog.completed ? 'Sudah Didiskusikan' : 'Belum Dikerjakan'}
                        </span>
                    </div>
                    <p class="text-xs text-slate-700 font-medium">
                        <strong>Soal HOTS:</strong> ${lvl.hots.story}
                    </p>
                `;
                detailsContainer.appendChild(item);
            });

            document.getElementById('reportCompletedLevels').innerText = `${completedCount} / 6`;
            document.getElementById('reportModal').classList.remove('hidden');
        }

        function hideReportModal() {
            playSound('click');
            document.getElementById('reportModal').classList.add('hidden');
        }
